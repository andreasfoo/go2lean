package levenshtein

import "unicode/utf8"

// Reading: documented to work on runes (Unicode code points), no normalization.

// go2lean:spec ComputeDistance
func SpecComputeDistanceLowerBound(a, b string, ret int) bool {
	d := utf8.RuneCountInString(a) - utf8.RuneCountInString(b)
	if d < 0 {
		d = -d
	}
	return ret >= d
}

// go2lean:spec ComputeDistance
func SpecComputeDistanceUpperBound(a, b string, ret int) bool {
	m := utf8.RuneCountInString(a)
	n := utf8.RuneCountInString(b)
	if n > m {
		m = n
	}
	return ret <= m
}

// go2lean:spec ComputeDistance
func SpecComputeDistanceEqualIsZero(a, b string, ret int) bool {
	if a != b {
		return true
	}
	return ret == 0
}

// go2lean:spec ComputeDistance
// Restricted to valid UTF-8: distinct invalid bytes may decode to the same rune.
func SpecComputeDistanceZeroOnlyIfEqual(a, b string, ret int) bool {
	if !utf8.ValidString(a) || !utf8.ValidString(b) {
		return true
	}
	if ret != 0 {
		return true
	}
	return a == b
}

// go2lean:spec ComputeDistance
func SpecComputeDistanceEmpty(a, b string, ret int) bool {
	if a == "" {
		return ret == utf8.RuneCountInString(b)
	}
	if b == "" {
		return ret == utf8.RuneCountInString(a)
	}
	return true
}

// go2lean:spec ComputeDistance
func SpecComputeDistanceHammingBound(a, b string, ret int) bool {
	if !utf8.ValidString(a) || !utf8.ValidString(b) {
		return true
	}
	if utf8.RuneCountInString(a) != utf8.RuneCountInString(b) {
		return true
	}
	i, j, diff := 0, 0, 0
	for i < len(a) && j < len(b) {
		r1, n1 := utf8.DecodeRuneInString(a[i:])
		r2, n2 := utf8.DecodeRuneInString(b[j:])
		if r1 != r2 {
			diff++
		}
		i += n1
		j += n2
	}
	return ret <= diff
}
