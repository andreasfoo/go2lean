package fuzzy

import "unicode/utf8"

// Reading: the docs speak of "characters"; a character is taken to be a rune
// (Unicode code point). Specs that compare characters are restricted to valid
// UTF-8 inputs and return true otherwise. The *Fold specs are restricted to
// ASCII inputs, where "case-insensitive" is unambiguous.

// go2lean:spec Match
func SpecMatchOnlyIfSubsequence(source, target string, ret bool) bool {
	if !ret {
		return true
	}
	if !utf8.ValidString(source) || !utf8.ValidString(target) {
		return true
	}
	j := 0
	for _, r := range source {
		found := false
		for j < len(target) {
			tr, size := utf8.DecodeRuneInString(target[j:])
			j += size
			if tr == r {
				found = true
				break
			}
		}
		if !found {
			return false
		}
	}
	return true
}

// go2lean:spec Match
func SpecMatchLength(source, target string, ret bool) bool {
	if !ret {
		return true
	}
	if !utf8.ValidString(source) || !utf8.ValidString(target) {
		return true
	}
	return utf8.RuneCountInString(source) <= utf8.RuneCountInString(target)
}

// go2lean:spec MatchFold
func SpecMatchFoldOnlyIfFoldedSubsequenceASCII(source, target string, ret bool) bool {
	if !ret {
		return true
	}
	for i := 0; i < len(source); i++ {
		if source[i] >= 0x80 {
			return true
		}
	}
	for i := 0; i < len(target); i++ {
		if target[i] >= 0x80 {
			return true
		}
	}
	j := 0
	for i := 0; i < len(source); i++ {
		c := source[i]
		if c >= 'A' && c <= 'Z' {
			c += 'a' - 'A'
		}
		found := false
		for j < len(target) {
			d := target[j]
			j++
			if d >= 'A' && d <= 'Z' {
				d += 'a' - 'A'
			}
			if c == d {
				found = true
				break
			}
		}
		if !found {
			return false
		}
	}
	return true
}

// go2lean:spec RankMatch
func SpecRankMatchRange(source, target string, ret int) bool {
	return ret >= -1
}

// go2lean:spec RankMatch
// When source is a subsequence of target, the Levenshtein distance is exactly
// the number of deleted characters: runes(target) - runes(source).
func SpecRankMatchValueIsDeletionCount(source, target string, ret int) bool {
	if ret == -1 {
		return true
	}
	if !utf8.ValidString(source) || !utf8.ValidString(target) {
		return true
	}
	return ret == utf8.RuneCountInString(target)-utf8.RuneCountInString(source)
}

// go2lean:spec RankMatchFold
func SpecRankMatchFoldRange(source, target string, ret int) bool {
	return ret >= -1
}

// go2lean:spec RankMatchFold
func SpecRankMatchFoldASCII(source, target string, ret int) bool {
	if ret == -1 {
		return true
	}
	for i := 0; i < len(source); i++ {
		if source[i] >= 0x80 {
			return true
		}
	}
	for i := 0; i < len(target); i++ {
		if target[i] >= 0x80 {
			return true
		}
	}
	j := 0
	for i := 0; i < len(source); i++ {
		c := source[i]
		if c >= 'A' && c <= 'Z' {
			c += 'a' - 'A'
		}
		found := false
		for j < len(target) {
			d := target[j]
			j++
			if d >= 'A' && d <= 'Z' {
				d += 'a' - 'A'
			}
			if c == d {
				found = true
				break
			}
		}
		if !found {
			return false
		}
	}
	return ret == len(target)-len(source)
}

// go2lean:spec LevenshteinDistance
func SpecLevenshteinDistanceLowerBound(s, t string, ret int) bool {
	d := utf8.RuneCountInString(s) - utf8.RuneCountInString(t)
	if d < 0 {
		d = -d
	}
	return ret >= d
}

// go2lean:spec LevenshteinDistance
func SpecLevenshteinDistanceUpperBound(s, t string, ret int) bool {
	m := utf8.RuneCountInString(s)
	n := utf8.RuneCountInString(t)
	if n > m {
		m = n
	}
	return ret <= m
}

// go2lean:spec LevenshteinDistance
func SpecLevenshteinDistanceEqualIsZero(s, t string, ret int) bool {
	if s != t {
		return true
	}
	return ret == 0
}

// go2lean:spec LevenshteinDistance
func SpecLevenshteinDistanceZeroOnlyIfEqual(s, t string, ret int) bool {
	if !utf8.ValidString(s) || !utf8.ValidString(t) {
		return true
	}
	if ret != 0 {
		return true
	}
	return s == t
}

// go2lean:spec LevenshteinDistance
func SpecLevenshteinDistanceEmpty(s, t string, ret int) bool {
	if s == "" {
		return ret == utf8.RuneCountInString(t)
	}
	if t == "" {
		return ret == utf8.RuneCountInString(s)
	}
	return true
}

// go2lean:spec LevenshteinDistance
// Equal rune length: substituting each differing position is one edit script.
func SpecLevenshteinDistanceHammingBound(s, t string, ret int) bool {
	if !utf8.ValidString(s) || !utf8.ValidString(t) {
		return true
	}
	if utf8.RuneCountInString(s) != utf8.RuneCountInString(t) {
		return true
	}
	i, j, diff := 0, 0, 0
	for i < len(s) && j < len(t) {
		r1, n1 := utf8.DecodeRuneInString(s[i:])
		r2, n2 := utf8.DecodeRuneInString(t[j:])
		if r1 != r2 {
			diff++
		}
		i += n1
		j += n2
	}
	return ret <= diff
}
