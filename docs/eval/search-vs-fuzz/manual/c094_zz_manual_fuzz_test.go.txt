package shellquote

import (
	"strings"
	"testing"
)

// Evaluation 5, arm B by hand: go2lean fuzz skips variadic targets. The
// arguments are the fuzzer's string split at zero bytes.
func FuzzManualJoin(f *testing.F) {
	f.Fuzz(func(t *testing.T, raw string) {
		args := strings.Split(raw, "\x00")
		if raw == "" {
			args = nil
		}
		if !SpecJoinRoundTrip(args, Join(args...)) {
			t.Fatalf("SpecJoinRoundTrip is false")
		}
	})
}
