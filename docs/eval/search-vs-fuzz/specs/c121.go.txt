package purell

import (
	"strconv"
	"strings"
)

// go2lean:sequence NormalizeURLString
// SeqRemoveDefaultPort: "NormalizeURLString returns the normalized string
// ... It takes an URL string as input, as well as the normalization flags",
// with FlagRemoveDefaultPort. Reference: RFC 3986 section 6.2.3,
// scheme-based normalization: "the port component ... should be omitted
// when its value is the default" (80 for http, 443 for https), and "an
// empty port ... should be normalized to omit the ':'". Other ports stay.
func SeqRemoveDefaultPort(host, path string, port int, https bool) bool {
	_ = specSeeds()
	scheme, def := "http", 80
	if https {
		scheme, def = "https", 443
	}
	host = pWord(host, "example") + ".com"
	path = "/" + pWord(path, "")
	ports := []int{80, 443, 8080, -1, def}
	p := ports[(port%5+5)%5]
	var in, want string
	switch {
	case p == -1: // empty port
		in, want = scheme+"://"+host+":"+path, scheme+"://"+host+path
	case p == def:
		in, want = scheme+"://"+host+":"+strconv.Itoa(p)+path, scheme+"://"+host+path
	default:
		in = scheme + "://" + host + ":" + strconv.Itoa(p) + path
		want = in
	}
	got, err := NormalizeURLString(in, FlagRemoveDefaultPort)
	return err == nil && got == want
}

// specSeeds are literals for the search.
func specSeeds() []string { return []string{"www", "a", "index", "x80", "80"} }

// pWord keeps the lowercase letters and digits of s (at most 6).
func pWord(s, fallback string) string {
	var b strings.Builder
	for _, r := range s {
		if (r >= 'a' && r <= 'z') || (r >= '0' && r <= '9') {
			b.WriteRune(r)
			if b.Len() == 6 {
				break
			}
		}
	}
	if b.Len() == 0 {
		return fallback
	}
	return b.String()
}
