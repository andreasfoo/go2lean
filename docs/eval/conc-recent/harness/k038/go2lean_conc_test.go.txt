package objstorageprovider

import (
	"context"
	"sync"

	"github.com/cockroachdb/pebble/internal/base"
	"github.com/cockroachdb/pebble/objstorage"
	"github.com/cockroachdb/pebble/objstorage/remote"
	"github.com/cockroachdb/pebble/vfs"
)

// go2lean:sequence provider.Sync
// SeqConcProviderSync: two goroutines create, sync and remove distinct objects on one Provider; Sync persists pending changes and must not fail because another goroutine is creating, removing or syncing other objects at the same time.
func SeqConcProviderSync(ops []byte, shared bool) bool {
	if len(ops) > 16 {
		ops = ops[:16]
	}
	st := DefaultSettings(vfs.NewMem(), "")
	st.Remote.StorageFactory = remote.MakeSimpleFactory(map[remote.Locator]remote.Storage{
		"": remote.NewInMem(),
	})
	st.Remote.CreateOnShared = true
	st.Remote.CreateOnSharedLocator = ""
	p, err := Open(st)
	if err != nil {
		return true
	}
	defer p.Close()
	if err := p.SetCreatorID(1); err != nil {
		return true
	}

	var mu sync.Mutex
	ok := true
	fail := func() {
		mu.Lock()
		ok = false
		mu.Unlock()
	}
	run := func(g int) {
		for i := g; i < len(ops); i += 2 {
			num := base.FileNum(100*g + i + 1).DiskFileNum()
			w, _, err := p.Create(context.Background(), base.FileTypeTable, num, objstorage.CreateOptions{
				PreferSharedStorage: shared,
			})
			if err != nil {
				fail()
				return
			}
			if err := w.Finish(); err != nil {
				fail()
				return
			}
			if ops[i]&1 != 0 {
				if err := p.Sync(); err != nil {
					fail()
					return
				}
			}
			if err := p.Remove(base.FileTypeTable, num); err != nil {
				fail()
				return
			}
			if ops[i]&2 != 0 {
				if err := p.Sync(); err != nil {
					fail()
					return
				}
			}
		}
	}

	var wg sync.WaitGroup
	wg.Add(2)
	go func() { defer wg.Done(); run(0) }()
	go func() { defer wg.Done(); run(1) }()
	wg.Wait()
	return ok
}
