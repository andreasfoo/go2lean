package amqp091

import (
	"bufio"
	"bytes"
	"errors"
	"net"
	"sync"
	"sync/atomic"
	"time"
)

type g2lRWC struct{}

func (g2lRWC) Read(p []byte) (int, error)  { return 0, nil }
func (g2lRWC) Write(p []byte) (int, error) { return len(p), nil }
func (g2lRWC) Close() error                { return nil }

// go2lean:sequence Connection.Reconnect
// SeqConcReconnectClose: an OnConnectionClose handler calls Reconnect on a
// dropped connection while the application calls Close. Close's doc: "After
// returning from this call, all resources associated with this connection ...
// will also be closed", so a Reconnect that starts after Close returned must
// report ErrClosed instead of rebuilding the connection; and whichever call
// wins, Close must not be left with a connection that Reconnect resurrects.
func SeqConcReconnectClose(retries uint8, open bool, deadline bool) bool {
	var buf bytes.Buffer
	conn := &Connection{
		conn:      g2lRWC{},
		writer:    &writer{bufio.NewWriter(&buf)},
		rpc:       make(chan message, 2),
		close:     make(chan struct{}),
		errors:    make(chan *Error, 1),
		sends:     make(chan time.Time, 100),
		deadlines: make(chan readDeadliner, 100),
		lifeCycle: newLifeCycle(),
		url:       "amqp://guest:guest@localhost:5672/",
		Config: Config{
			Recovery: &Recovery{
				ReconnectionConfig: &ReconnectionConfig{
					MaxRetryCount: int(retries%2) + 1,
					RetryInterval: time.Millisecond,
				},
			},
			Dial: func(network, addr string) (net.Conn, error) {
				return nil, errors.New("broker unreachable")
			},
		},
	}
	conn.lifeCycle.SetState(StateOpen, nil)
	if open {
		// healthy connection: the broker answers connection.close
		conn.rpc <- &connectionCloseOk{}
	} else {
		// dropped connection, recovery pending
		conn.closed.Store(true)
	}

	var closeReturned atomic.Bool
	ok := true
	var wg sync.WaitGroup
	wg.Add(2)
	go func() {
		defer wg.Done()
		after := closeReturned.Load()
		err := conn.Reconnect()
		if after && err != ErrClosed {
			ok = false
		}
	}()
	go func() {
		defer wg.Done()
		if deadline {
			conn.CloseDeadline(time.Now().Add(time.Second))
		} else {
			conn.Close()
		}
		closeReturned.Store(true)
	}()
	wg.Wait()
	return ok
}
