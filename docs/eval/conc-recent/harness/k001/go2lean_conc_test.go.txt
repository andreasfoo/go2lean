package transport

import "sync"

// go2lean:sequence bdpEstimator.timesnap
// SeqConcBDPTimesnap: the transport's writer goroutine records the send time of
// the BDP ping (timesnap) while the reader goroutine accounts received bytes
// (add) and handles the ping ack (calculate) on the same estimator; the fix says
// "fix data race on bdpEstimator.sentAt".
func SeqConcBDPTimesnap(n uint32, rounds uint8, ackOther bool) bool {
	b := &bdpEstimator{
		bdp:               initialWindowSize,
		updateFlowControl: func(uint32) {},
	}
	k := int(rounds%3) + 1
	d := bdpPing.data
	if ackOther {
		d[0]++
	}
	var wg sync.WaitGroup
	wg.Add(2)
	go func() {
		defer wg.Done()
		for i := 0; i < k; i++ {
			b.timesnap(bdpPing.data)
		}
	}()
	go func() {
		defer wg.Done()
		for i := 0; i < k; i++ {
			b.add(n)
			b.calculate(d)
		}
	}()
	wg.Wait()
	return true
}
